# 饥荒工业级最佳实践范式：附魔拓展补丁
> 对应工坊权威基准：Mod 3605108247 (`附魔拓展补丁`)
> 技术分类：05_战斗AI行为树与Boss机制类 | 源码规模：17 文件 / 2672 行
本范式旨在解决饥荒联机版中 附魔拓展补丁 涉及的架构协同难题。
在官方源码机制下，原版逻辑对相关切面具备较强的封闭性。通过精准挂载钩子、隔离脏标记通信及遵循生命周期规范，实现无侵入式工业级扩展。

## 二、核心底层算法与网络拓扑机制
1. **系统注入切面与挂载拓扑**：
   - 提取到核心切面钩子：`['AddPrefabPostInit:killerbee', 'AddPrefabPostInit:world', 'AddComponentPostInit:stewer']`；
   - 依赖官方底层实体生命周期：在 `PreInit` 完成网络数据对齐，在 `PostInit` 挂载业务监听，确保跨客户端无缝通信。

```lua
local GLOBAL = GLOBAL

AddPrefabPostInit("killerbee", function(inst)

    if not GLOBAL.TheNet:GetIsServer() then return end
    
    local config_value = GLOBAL.KillerBeeEnchantDropsEnabled
    if config_value == nil then
        config_value = GetModConfigData("ENABLE_KILLERBEE_ENCHANT_DROPS") or false
        GLOBAL.KillerBeeEnchantDropsEnabled = config_value
    end

    if not inst.components.lootdropper then
        return
    end
    
    print("[KillerBee Event] 加载杀人蜂事件过滤...")
    
    local function GetConfig()
        if GetModConfigData then
            local success, result = pcall(GetModConfigData, "ENABLE_KILLERBEE_ENCHANT_DROPS")
            if success then
                return result
            end
        end
        return false
    end
    
    local config_value = GetConfig()
    print("[KillerBee Event] 杀人蜂配置: " .. (config_value and "允许掉落" or "禁止掉落"))
    
    if config_value then
        return
    end
    
    local FILTER_ITEMS = {
        "hh_effect_stone",
        "hh_essence",
        "hh_remove_stone", 
        "hh_effect_tally",
    }
    
    inst:ListenForEvent("spawnloot", function(inst, data)
        if not data or not data.item then
            return
        end
        
        local item_prefab = data.item.prefab
        for _, filter_item in ipairs(FILTER_ITEMS) do
            if item_prefab == filter_item then
                print("[KillerBee Event] 🚫 销毁杀人蜂掉落的附魔石: " .. tostring(item_prefab))
                data.item:Remove()
                break
            end
        end
end)
    
    print("[KillerBee Event] 杀人蜂事件过滤加载完成")
end)
```

### 变体演进：Keep Spiders When Despawning (Mod 2529730952)
- **变体场景与切面**：收录来自工坊 Mod 2529730952 (`Keep Spiders When Despawning`) 的生产级切面。
- **核心切面代码**：
```lua
-- ============================================================================
-- 生产级架构复刻切面范式 (提取自 Mod 2529730952)
-- ============================================================================
local GLOBAL = GLOBAL

-- 切面钩子：AddPrefabPostInit:webber
AddPrefabPostInit("webber", function(inst)
	if not GLOBAL.TheWorld.ismastersim then
		return
	end
	inst.webber_followers = {}
	OnDespawn_old = inst.OnDespawn
	inst.OnDespawn = function(inst, migrationdata, ...)
		for k, v in pairs(inst.components.leader.followers) do
			if k:HasTag("spider") and (k.components.inventoryitem == nil or not (k.components.inventoryitem:GetGrandOwner() == inst)) then
				local savedata = k:GetSaveRecord()
				table.insert(inst.webber_followers, savedata)
				--remove spiders
				k:AddTag("notarget")
				k:AddTag("NOCLICK")
				k.persists = false
				if k.components.health then
					k.components.health:SetInvincible(true)
				end
				k:DoTaskInTime(math.random()*0.2, function(k)
					local fx = GLOBAL.SpawnPrefab("spawn_fx_small")
					fx.Transform:SetPosition(k.Transform:GetWorldPosition())
					if not k.components.colourtweener then
						k:AddComponent("colourtweener")
					end
					k.components.colourtweener:StartTween({ 0, 0, 0, 1 }, 13 * GLOBAL.FRAMES, k.Remove)
				end)
			end
		end
		return OnDespawn_old(inst, migrationdata, ...)
	end
	
	OnSave_old = inst.OnSave
	inst.OnSave = function(inst, data, ...)
		data.webber_followers = inst.webber_followers
		if OnSave_old ~= nil then
			return OnSave_old(inst, data, ...)
		end
	end
	
	OnLoad_old = inst.OnLoad
	inst.OnLoad = function(inst, data, ...)
		if data and data.webber_followers then
			for k, v in pairs(data.webber_followers) do
				inst:DoTaskInTime(0.2*math.random(), function(inst)
					local spider = GLOBAL.SpawnSaveRecord(v)
					inst.components.leader:AddFollower(spider)
					spider:DoTaskInTime(0, function(spider)
						if inst:IsValid() and not spider:IsNear(inst, 8) then -- TODO
							spider.Transform:SetPosition(inst.Transform:GetWorldPosition())
							spider.sg:GoToState("idle")
						end
					end)
					local fx = GLOBAL.SpawnPrefab("spawn_fx_small")
					fx.Transform:SetPosition(spider.Transform:GetWorldPosition())
				end)
			end
		end
		if OnLoad_old ~= nil then
			return OnLoad_old(inst, data, ...)
		end
	end
end)
```

## 六、踩坑与反模式避错指南（防御性闭环）
1. **网络变量声明时机**：所有 `net_*` 必须在 `inst.entity:AddNetwork()` 后、`SetPristine()` 前注册，严禁在预制体后处理的尾部动态挂载。
2. **重入锁防御**：在复杂事件分发时，务必维护实体级别的布尔标记，防止死循环递归。
