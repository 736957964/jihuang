# 饥荒工业级最佳实践范式：附魔拓展补丁
> 对应工坊权威基准：Mod 3605108247 (`附魔拓展补丁`)
> 技术分类：05_战斗AI行为树与Boss机制类 | 源码规模：17 文件 / 2672 行
本范式旨在解决饥荒联机版中 附魔拓展补丁 涉及的架构协同难题。
在官方源码机制下，原版逻辑对相关切面具备较强的封闭性。通过精准挂载钩子、隔离脏标记通信及遵循生命周期规范，实现无侵入式工业级扩展。

## 二、核心底层算法与网络拓扑机制
1. **系统注入切面与挂载拓扑**：
   - 提取到核心切面钩子：`['AddPrefabPostInit:killerbee', 'AddPrefabPostInit:world', 'AddComponentPostInit:stewer']`；
   - 依赖官方底层实体生命周期：在 `PreInit` 完成网络数据对齐，在 `PostInit` 挂载业务监听，确保跨客户端无缝通信。

```lua
local GLOBAL = GLOBAL

AddPrefabPostInit("killerbee", function(inst)

  if not GLOBAL.TheNet:GetIsServer() then return end
  
  local config_value = GLOBAL.KillerBeeEnchantDropsEnabled
  if config_value == nil then
      config_value = GetModConfigData("ENABLE_KILLERBEE_ENCHANT_DROPS") or false
      GLOBAL.KillerBeeEnchantDropsEnabled = config_value
  end

  if not inst.components.lootdropper then
      return
  end
  
  print("[KillerBee Event] 加载杀人蜂事件过滤...")
  
  local function GetConfig()
      if GetModConfigData then
          local success, result = pcall(GetModConfigData, "ENABLE_KILLERBEE_ENCHANT_DROPS")
          if success then
              return result
          end
      end
      return false
  end
  
  local config_value = GetConfig()
  print("[KillerBee Event] 杀人蜂配置: " .. (config_value and "允许掉落" or "禁止掉落"))
  
  if config_value then
      return
  end
  
  local FILTER_ITEMS = {
      "hh_effect_stone",
      "hh_essence",
      "hh_remove_stone", 
      "hh_effect_tally",
  }
  
  inst:ListenForEvent("spawnloot", function(inst, data)
      if not data or not data.item then
          return
      end
      
      local item_prefab = data.item.prefab
      for _, filter_item in ipairs(FILTER_ITEMS) do
          if item_prefab == filter_item then
              print("[KillerBee Event] 🚫 销毁杀人蜂掉落的附魔石: " .. tostring(item_prefab))
              data.item:Remove()
              break
          end
      end
end)
  
  print("[KillerBee Event] 杀人蜂事件过滤加载完成")
end)
```

## 四、方案对比矩阵与权衡取舍（Trade-off Matrix）

| 技术选型方案 | 核心技术原理 | 优势与收益 | 代价与潜在风险 | 推荐适用场景 |
| :--- | :--- | :--- | :--- | :--- |
| **方案 A：PostInit 深度切面（本母本）** | 挂载官方标准生命周期注入点 | 原生兼容性高、零侵入、支持跨 Mod 协同 | 需精确处理加载先后时序 | 绝大多数公网联机与模组服 |
| **方案 B：全局暴力覆写（Anti-Pattern）** | 篡改全局函数或暴力替换 Class | 编写简单快速、无需排查依赖链 | 极易造成 Mod 相互排斥及坏档 | 个人单机测试环境 |

```lua
		return
	end
	inst.webber_followers = {}
	OnDespawn_old = inst.OnDespawn
	inst.OnDespawn = function(inst, migrationdata, ...)
		for k, v in pairs(inst.components.leader.followers) do
			if k:HasTag("spider") and (k.components.inventoryitem == nil or not (k.components.inventoryitem:GetGrandOwner() == inst)) then
				local savedata = k:GetSaveRecord()
				table.insert(inst.webber_followers, savedata)
				--remove spiders
				k:AddTag("notarget")
				k:AddTag("NOCLICK")
				k.persists = false
				if k.components.health then
					k.components.health:SetInvincible(true)
				end
				k:DoTaskInTime(math.random()*0.2, function(k)
					local fx = GLOBAL.SpawnPrefab("spawn_fx_small")
					fx.Transform:SetPosition(k.Transform:GetWorldPosition())
					if not k.components.colourtweener then
						k:AddComponent("colourtweener")
					end
					k.components.colourtweener:StartTween({ 0, 0, 0, 1 }, 13 * GLOBAL.FRAMES, k.Remove)
				end)
			end
		end
		return OnDespawn_old(inst, migrationdata, ...)
	end
	
	OnSave_old = inst.OnSave
	inst.OnSave = function(inst, data, ...)
		data.webber_followers = inst.webber_followers
		if OnSave_old ~= nil then
			return OnSave_old(inst, data, ...)
		end
	end
	
	OnLoad_old = inst.OnLoad
	inst.OnLoad = function(inst, data, ...)
		if data and data.webber_followers then
			for k, v in pairs(data.webber_followers) do
				inst:DoTaskInTime(0.2*math.random(), function(inst)
					local spider = GLOBAL.SpawnSaveRecord(v)

					end)
					local fx = GLOBAL.SpawnPrefab("spawn_fx_small")
					fx.Transform:SetPosition(spider.Transform:GetWorldPosition())
				end)
			end
		end
		if OnLoad_old ~= nil then
			return OnLoad_old(inst, data, ...)
		end
```

```lua
local GLOBAL = GLOBAL

-- 切面钩子：AddComponentPostInit:health
AddComponentPostInit("health", function(self)
	self.scaling = {}
	self.scaling.factor = SCALING_RATIO
	self.scaling.baseplayers = 1
	self.scaling.multiplier = 1
	self.scaling.active = false

	-- Enables Health Scaling
	function self:EnableScaling()
		self.scaling.basehealth = self.maxhealth
		self.scaling.active = true
		self.scaling.task = self.inst:DoPeriodicTask(SCALING_PERIOD, ScaleHealth(self))
	end

	-- Disables Health Scaling
	function self:DisableScaling()
		if self.scaling.active then
			self.scaling.task:Cancel()
			self.scaling.active = false

			local currentHealthPercent = self:GetPercent()
			self:SetMaxHealth(self.scaling.basehealth)
			self:SetPercent(currentHealthPercent)
		end
	end

	local _SetMaxHealth = self.SetMaxHealth
	function self:SetMaxHealth(amount, isHealthScaling)
		if not isHealthScaling and self.scaling.active then
			-- Health changed externally while we're scaling, update the base health instead
			self.scaling.basehealth = amount
			-- Then recalculate
			ScaleHealth(self)
		else
			_SetMaxHealth(self, amount)
		end
	end

	local _OnSave = self.OnSave
	function self:OnSave(...)
		data = _OnSave(self, ...)
		if self.scaling.active then
			data.healthpercent = self:GetPercent()
		end

		return data
	end

	local _OnLoad = self.OnLoad
	function self:OnLoad(data, ...)
		_OnLoad(self, data, ...)

		if self.scaling.active and data.healthpercent then
			self:SetPercent(data.healthpercent)
		end
	end
end)
```

```lua
  -- 1. 保留原有的 GetAttacked 覆盖（火焰减免 + 位面防御）
  local old_GetAttacked = Combat.GetAttacked
  
  function Combat:GetAttacked(attacker, damage, weapon, stimuli, ...)
      -- 处理火焰伤害减免
      if self.inst:HasTag("fire_resistant") and damage > 0 then
          local is_fire_damage = (weapon and weapon:HasTag("fire")) or 
                                (attacker and attacker:HasTag("fire")) or
                                (stimuli == "fire")
          
          if is_fire_damage then
              local best_resistance = 0
              if self.inst.components.inventory then
                  for _, item in pairs(self.inst.components.inventory.equipslots) do
                      if item and fire_heat_resistant_gear[item.prefab] then
                          local resistance = fire_heat_resistant_gear[item.prefab].fire_resistance
                          if resistance > best_resistance then
                              best_resistance = resistance
                          end
                      end
                  end
              end
              
              damage = damage * (1 - best_resistance)
          end
      end
      
      -- 处理位面伤害和位面防御
      if damage > 0 then
          local is_planar_damage = (weapon and weapon:HasTag("planar")) or 
                                  (attacker and attacker:HasTag("planar")) or
                                  (stimuli and stimuli == "planar")
          
          if is_planar_damage then
              local total_planar_defense = 0
              if self.inst.components.inventory then
                  for _, item in pairs(self.inst.components.inventory.equipslots) do
                      if item and item.components.armor and item.components.armor.planardefense then
                          total_planar_defense = total_planar_defense + item.components.armor.planardefense
                      end
                  end
              end
              
              damage = math.max(0, damage - total_planar_defense)
          end
      end
      
      return old_GetAttacked(self, attacker, damage, weapon, stimuli, ...)
  end

  -- 2. 【增强版】覆盖 Zg_GetAttacked，实现蝴蝶电击完全免疫（通过伤害类型和攻击者标签双重判断）
  local old_ZgGetAttacked = Combat.Zg_GetAttacked
  function Combat:Zg_GetAttacked(attacker, damage, weapon, stimuli, ...)
      -- 如果受伤者拥有 "zg_ele_immune" 标签
      if self.inst:HasTag("zg_ele_immune") then
          -- 判断是否为蝴蝶电击： stimuli 为 "zg_ele_electric" 或 攻击者带有 "zg_butterfly" 标签
          local is_butterfly_damage = (stimuli == "zg_ele_electric") or 
                                      (attacker and attacker:HasTag("zg_butterfly"))
          if is_butterfly_damage then
              -- 完全忽略该次伤害，不执行任何后续操作
              return
          end
      end
      -- 其他情况调用原函数（如果存在），否则不做任何事
      if old_ZgGetAttacked then
          return old_ZgGetAttacked(self, attacker, damage, weapon, stimuli, ...)
      end
      -- 如果没有旧函数，则忽略（容错）
  end
end)
```
