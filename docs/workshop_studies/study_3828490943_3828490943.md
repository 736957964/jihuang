### AddComponentPostInit("weapon")
```lua

AddComponentPostInit("weapon", function(self)
    local old_launch = self.LaunchProjectile
    if old_launch == nil then
        return
    end

    self.LaunchProjectile = function(self_, attacker, target)
        local inst = self_.inst
        if inst == nil or not inst:HasTag("slingshot") then
            return old_launch(self_, attacker, target)
        end
```
