-- 客户端图标：物品没有自己的 tex，直接本地覆盖成清洁扫帚的图标，
-- 避免客户端去 inventoryimages 里找 boatreskinbrush.tex 报警告。
AddPrefabPostInit("boatreskinbrush", function(inst)
    if GLOBAL.TheWorld ~= nil and not GLOBAL.TheWorld.ismastersim then
        inst:DoTaskInTime(0, function(item)
            if item:IsValid() and item.replica ~= nil and item.replica.inventoryitem ~= nil then
                item.replica.inventoryitem:OverrideImage("reskin_tool")
            end
