## 三、关键代码片段

### AddComponentPostInit("builder")
```lua

AddComponentPostInit("builder", function(self)
    local old_DoBuild = self.DoBuild
    function self:DoBuild(recname, pt, rotation, skin)
        local need = UPGRADE_MAP[recname]
        if need ~= nil and self.inst.components.inventory ~= nil then
            local oldknife = self.inst.components.inventory:FindItem(function(item)
                return item.prefab == need and item.components.xqg_glassmark ~= nil
            end)
            if oldknife ~= nil then
                self.inst._xqg_pending_stars = oldknife.components.xqg_glassmark:GetBonus()
            end
```
